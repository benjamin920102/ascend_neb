cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(neb_int4 LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_ASC_ARCHITECTURES dav-2201 CACHE STRING "NPU architecture")
add_library(neb_int4 SHARED matmul_int4.asc bridge.cpp)
set_target_properties(neb_int4 PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(neb_int4 PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${CMAKE_ASC_ARCHITECTURES}>)
target_link_libraries(neb_int4 PRIVATE tiling_api register platform ascendcl m dl pthread)
